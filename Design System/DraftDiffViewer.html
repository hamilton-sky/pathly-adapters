export function buildSendPrompt(
  filePath: string,
  fileBody: string,
  comments: Comment[]
): string {
  const norm = filePath.replace(/\\/g, '/')
  const joined = comments
    .map(c => `[line ${c.line}] ${c.text}`)
    .join('\n')
  return [
    `## Task`,
    `Fix the following comments in ${norm}:`,
    ``,
    joined,
    ``,
    `## File content`,
    fileBody,
  ].join('\n')
}

export function getSpawnCwd(filePath: string): string {
  const norm = filePath.replace(/\\/g, '/')
  const parts = norm.split('/')
  const studioIdx = parts.lastIndexOf('studio')
  return studioIdx >= 0
    ? parts.slice(0, studioIdx + 1).join('/')
    : path.dirname(filePath)
}
